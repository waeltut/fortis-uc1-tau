cmake_minimum_required(VERSION 3.19)
project(mur_base_selector)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(ament_cmake REQUIRED)
find_package(ament_cmake_python REQUIRED)
foreach(dep rclcpp moveit_core moveit_ros_planning)
  find_package(${dep} REQUIRED)
endforeach()
find_package(yaml-cpp REQUIRED)
find_package(Python3 COMPONENTS Interpreter Development REQUIRED)
find_package(pybind11_vendor REQUIRED)
find_package(pybind11 REQUIRED)
ament_get_python_install_dir(PYTHON_INSTALL_DIR)
pybind11_add_module(mur_base_selector_engine NO_EXTRAS src/offline_engine.cpp)
ament_target_dependencies(mur_base_selector_engine PUBLIC rclcpp moveit_core moveit_ros_planning)
target_link_libraries(mur_base_selector_engine PRIVATE yaml-cpp)
target_compile_options(mur_base_selector_engine PRIVATE -Wall -Wextra -Wpedantic)
install(TARGETS mur_base_selector_engine LIBRARY DESTINATION "${PYTHON_INSTALL_DIR}")
ament_environment_hooks("${CMAKE_CURRENT_SOURCE_DIR}/env-hooks/pythonpath.dsv.in")
ament_python_install_package(${PROJECT_NAME})
# ZIP extraction may discard source execute bits. Under --symlink-install,
# PROGRAMS can link directly to those non-executable files. Prepare executable
# copies in the build tree so installation does not rely on source permissions.
file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/launchers")
foreach(launcher base_selector build_volume volume_visualizer)
  set(launcher_path "${CMAKE_CURRENT_BINARY_DIR}/launchers/${launcher}")
  configure_file("${CMAKE_CURRENT_SOURCE_DIR}/scripts/${launcher}" "${launcher_path}" COPYONLY)
  file(CHMOD "${launcher_path}" PERMISSIONS
    OWNER_READ OWNER_WRITE OWNER_EXECUTE GROUP_READ GROUP_EXECUTE WORLD_READ WORLD_EXECUTE)
  install(PROGRAMS "${launcher_path}" DESTINATION lib/${PROJECT_NAME})
endforeach()
install(DIRECTORY config launch docs DESTINATION share/${PROJECT_NAME} PATTERN "*.npz" EXCLUDE)
install(FILES README.md DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)
  find_package(ament_cmake_pytest REQUIRED)
  ament_add_pytest_test(python_tests test TIMEOUT 120 APPEND_ENV PYTHONPATH=${CMAKE_CURRENT_SOURCE_DIR})
  find_package(ament_cmake_gtest REQUIRED)
  find_package(urdf REQUIRED)
  find_package(srdfdom REQUIRED)
  ament_add_gtest(test_arm_collision_policy test/test_arm_collision_policy.cpp)
  target_include_directories(test_arm_collision_policy PRIVATE src)
  ament_target_dependencies(test_arm_collision_policy moveit_core moveit_ros_planning urdf srdfdom)
endif()
ament_package()
