cmake_minimum_required(VERSION 3.16)
project(mur_reachability)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(ament_cmake REQUIRED)
find_package(ament_cmake_python REQUIRED)
foreach(dep rclcpp moveit_core moveit_ros_planning)
  find_package(${dep} REQUIRED)
endforeach()
find_package(yaml-cpp REQUIRED)
find_package(Python3 COMPONENTS Interpreter Development REQUIRED)
find_package(pybind11_vendor REQUIRED)
find_package(pybind11 REQUIRED)
ament_get_python_install_dir(PYTHON_INSTALL_DIR)
pybind11_add_module(mur_reachability_engine NO_EXTRAS src/offline_engine.cpp)
ament_target_dependencies(mur_reachability_engine PUBLIC rclcpp moveit_core moveit_ros_planning)
target_link_libraries(mur_reachability_engine PRIVATE yaml-cpp)
target_compile_options(mur_reachability_engine PRIVATE -Wall -Wextra -Wpedantic)
install(TARGETS mur_reachability_engine LIBRARY DESTINATION "${PYTHON_INSTALL_DIR}")
ament_environment_hooks("${CMAKE_CURRENT_SOURCE_DIR}/env-hooks/pythonpath.dsv.in")
install(DIRECTORY config DESTINATION share/${PROJECT_NAME})
install(FILES README.md DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  find_package(urdf REQUIRED)
  find_package(srdfdom REQUIRED)
  ament_add_gtest(test_arm_collision_policy test/test_arm_collision_policy.cpp)
  target_include_directories(test_arm_collision_policy PRIVATE src)
  ament_target_dependencies(test_arm_collision_policy moveit_core moveit_ros_planning urdf srdfdom)
endif()
ament_package()
