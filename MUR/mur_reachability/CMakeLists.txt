cmake_minimum_required(VERSION 3.16)
project(mur_reachability)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(ament_cmake REQUIRED)
find_package(rosidl_default_generators REQUIRED)
foreach(dep rclcpp std_msgs std_srvs geometry_msgs sensor_msgs visualization_msgs moveit_msgs moveit_core moveit_ros_planning tf2 tf2_ros)
  find_package(${dep} REQUIRED)
endforeach()
find_package(yaml-cpp REQUIRED)
rosidl_generate_interfaces(${PROJECT_NAME} "srv/FindBaseCandidates.srv" "msg/LayeredArea.msg" DEPENDENCIES std_msgs geometry_msgs sensor_msgs)
rosidl_get_typesupport_target(typesupport ${PROJECT_NAME} rosidl_typesupport_cpp)
foreach(target cache_server base_candidates)
  add_executable(${target} src/nodes.cpp)
  target_include_directories(${target} PRIVATE include)
  ament_target_dependencies(${target} rclcpp std_msgs std_srvs geometry_msgs sensor_msgs visualization_msgs moveit_msgs moveit_core moveit_ros_planning tf2 tf2_ros)
  target_link_libraries(${target} "${typesupport}" yaml-cpp)
  target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
endforeach()
target_compile_definitions(cache_server PRIVATE MR_CACHE_MODE=1)
install(TARGETS cache_server base_candidates DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY launch config DESTINATION share/${PROJECT_NAME})
install(FILES README.md DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)
  enable_testing()
  add_executable(core_test test/core_test.cpp)
  target_include_directories(core_test PRIVATE include)
  add_test(NAME core_test COMMAND core_test)
  add_executable(area_test test/area_test.cpp)
  target_include_directories(area_test PRIVATE include)
  add_test(NAME area_test COMMAND area_test)
  add_executable(refine_test test/refine_test.cpp)
  target_include_directories(refine_test PRIVATE include)
  add_test(NAME refine_test COMMAND refine_test)
endif()
ament_export_dependencies(rosidl_default_runtime)
ament_package()
